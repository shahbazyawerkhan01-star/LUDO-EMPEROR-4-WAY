import http from 'node:http'; import path from 'node:path'; import { fileURLToPath } from 'node:url'; import express from 'express'; import dotenv from 'dotenv'; import { WebSocketServer } from 'ws'; import { createClient } from '@supabase/supabase-js'; dotenv.config(); const app = express(); const port = Number(process.env.PORT || 8080); const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); const publicDir = path.join(__dirname, '..', 'public'); const supabase = createClient( process.env.SUPABASE_URL || '', process.env.SUPABASE_ANON_KEY || '' ); app.use(express.json()); app.use(express.static(publicDir)); const roomState = new Map(); function makeRoomCode(length = 4) { const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; let code = ''; for (let i = 0; i < length; i += 1) { code += chars[Math.floor(Math.random() * chars.length)]; } return code; } async function persistRoom(code, payload) { if (!process.env.SUPABASE_URL || !process.env.SUPABASE_ANON_KEY) return; try { const { data, error } = await supabase .from('rooms') .upsert({ code, ...payload, updated_at: new Date().toISOString() }) .select(); if (error) console.error('Supabase upsert error:', error.message); return data; } catch (err) { console.error('Supabase failed:', err.message); } } async function getRoomFromDb(code) { if (!process.env.SUPABASE_URL || !process.env.SUPABASE_ANON_KEY) return null; try { const { data, error } = await supabase .from('rooms') .select('*') .eq('code', code) .maybeSingle(); if (error) { console.error('Supabase room fetch error:', error.message); return null; } return data; } catch (err) { console.error('Supabase read error:', err.message); return null; } } function roomPayload(code, players = []) { return { code, players, state: roomState.get(code) || { phase: 'lobby', players: [] }, updated_at: new Date().toISOString(), }; } app.get('/api/health', (req, res) => { res.json({ ok: true, service: 'ludo-emperor-4-way-multiplayer', time: new Date().toISOString() }); }); app.post('/api/rooms/create', async (req, res) => { const { playerName = 'Player' } = req.body || {}; let code = makeRoomCode(); while (roomState.has(code)) { code = makeRoomCode(); } const payload = roomPayload(code, [{ name: playerName, connected: true }]); roomState.set(code, payload.state); await persistRoom(code, { players: payload.players, state: payload.state }); res.json({ ok: true, roomCode: code, shareUrl: `/?room=${code}` }); }); app.post('/api/rooms/join', async (req, res) => { const { roomCode, playerName = 'Player' } = req.body || {}; if (!roomCode) { return res.status(400).json({ ok: false, message: 'roomCode is required' }); } const normalized = String(roomCode).toUpperCase(); const existing = roomState.get(normalized) || await getRoomFromDb(normalized); if (!existing) { return res.status(404).json({ ok: false, message: 'Room not found' }); } const currentPlayers = Array.isArray(existing.players) ? existing.players : []; const player = { name: playerName, connected: true }; const updatedPlayers = [...currentPlayers.filter((p) => p.name !== playerName), player]; const state = typeof existing.state === 'object' ? existing.state : { phase: 'lobby', players: [] }; state.players = updatedPlayers; roomState.set(normalized, state); await persistRoom(normalized, { players: updatedPlayers, state }); res.json({ ok: true, roomCode: normalized, players: updatedPlayers, state }); }); app.get('/api/rooms/:code', async (req, res) => { const code = String(req.params.code).toUpperCase(); const room = roomState.get(code) || await getRoomFromDb(code); if (!room) { return res.status(404).json({ ok: false, message: 'Room not found' }); } res.json({ ok: true, roomCode: code, players: room.players || [], state: room.state || { phase: 'lobby', players: [] } }); }); const server = http.createServer(app); const wss = new WebSocketServer({ server }); const roomMembers = new Map(); wss.on('connection', (ws) => { let playerName = 'Player'; let roomCode = null; ws.on('message', (raw) => { try { const message = JSON.parse(raw.toString()); if (message.type === 'join-room') { roomCode = String(message.roomCode || '').toUpperCase(); playerName = message.playerName || playerName; if (!roomCode) { ws.send(JSON.stringify({ type: 'error', message: 'Room code is required' })); return; } if (!roomMembers.has(roomCode)) roomMembers.set(roomCode, new Set()); roomMembers.get(roomCode).add(ws); const room = roomState.get(roomCode) || { phase: 'lobby', players: [] }; room.players = Array.isArray(room.players) ? room.players : []; const existing = room.players.find((player) => player.name === playerName); if (!existing) { room.players.push({ name: playerName, connected: true }); } roomState.set(roomCode, room); persistRoom(roomCode, { players: room.players, state: room }); broadcast(roomCode, { type: 'room-update', roomCode, players: room.players, state: room }); return; } if (message.type === 'game-state' && roomCode) { const room = roomState.get(roomCode) || { phase: 'lobby', players: [] }; room.state = message.state || room.state || { phase: 'lobby', players: [] }; roomState.set(roomCode, room); persistRoom(roomCode, { players: room.players || [], state: room.state }); broadcast(roomCode, { type: 'game-state', roomCode, state: room.state }); return; } if (message.type === 'chat' && roomCode) { broadcast(roomCode, { type: 'chat', playerName, message: message.message }); } } catch (error) { console.error('WS message parse error:', error.message); } }); ws.on('close', () => { if (!roomCode) return; const members = roomMembers.get(roomCode); if (members) { members.delete(ws); if (members.size === 0) roomMembers.delete(roomCode); } const room = roomState.get(roomCode); if (room && Array.isArray(room.players)) { room.players = room.players.filter((player) => player.name !== playerName); roomState.set(roomCode, room); persistRoom(roomCode, { players: room.players, state: room.state || room }); broadcast(roomCode, { type: 'room-update', roomCode, players: room.players, state: room }); } }); }); function broadcast(roomCode, payload) { const members = roomMembers.get(roomCode) || new Set(); for (const client of members) { if (client.readyState === client.OPEN) { client.send(JSON.stringify(payload)); } } } server.listen(port, () => { console.log(`Ludo Emperor multiplayer server is running on http://localhost:${port}`); });
